country_event = {
	id = 1337010
	title = "State Seizure"
	desc = "Is $FROMCOUNTRY$ the country you want land from?"
	
	is_triggered_only = yes
	allow_multiple_instances = yes

	option = {
		name = "Yes."

		any_country = {
			limit = {
				vassal_of = THIS
				has_country_modifier = country_marker
			}
			remove_country_modifier = country_marker
		}

		random_owned = {
			limit = {
				THIS = {
					NOT = {has_country_modifier = vassal_state_seized}
				}
			}
		
			FROM = {
				random_owned = {
					limit = {
						NOT = {has_province_modifier = province_marker}
					}
					province_event = { id = 1337011 days = 0 }
				}
			}
		}
	}
	option = {
		name = "No."
		FROM = {
			add_country_modifier = {
				name = country_marker
				duration = 30
			}
		}
		random_country = {
			limit = {
				vassal_of = THIS
				NOT = {has_country_modifier = country_marker}
			}
			country_event = { id = 1337009 days = 0 }
		}
	}
}
